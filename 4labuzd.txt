FROM EMPLOYEE
INNER JOIN DEPARTMENT ON EMPLOYEE.DEPARTMENT_ID = DEPARTMENT.DEPARTMENT_ID

-- 3 Uzduotis
-- > 1500
-- sY+NVL -> EMPLOYEE.SALARY + NVL, speju, kad sY yra salary ;Dd
SELECT
    EMPLOYEE.FIRST_NAME,
    TO_CHAR(EMPLOYEE.HIRE_DATE, 'YYYY-MM-DD'),
    EMPLOYEE.DEPARTMENT_ID,
    DEPARTMENT.NAME,
    LOCATION.REGIONAL_GROUP
FROM EMPLOYEE
    INNER JOIN DEPARTMENT ON EMPLOYEE.DEPARTMENT_ID = DEPARTMENT.DEPARTMENT_ID
    INNER JOIN LOCATION ON DEPARTMENT.LOCATION_ID = LOCATION.LOCATION_ID
WHERE EMPLOYEE.SALARY + NVL(EMPLOYEE.COMMISSION, 0) > 1500;

-- 11 Uzdutois
SELECT
    JOB.FUNCTION
FROM EMPLOYEE
    INNER JOIN JOB ON EMPLOYEE.JOB_ID = JOB.JOB_ID
WHERE 
    EXTRACT(YEAR FROM HIRE_DATE) IN (1986, 1987) AND
    EXTRACT(MONTH FROM HIRE_DATE) BETWEEN 1 AND 6
GROUP BY JOB.FUNCTION;

-- 12 Uzduotis
-- pritruko order by
-- department id nepraso
-- visiems isskyrus klerkus
SELECT
    EMPLOYEE.FIRST_NAME,
    JOB.FUNCTION,
    EMPLOYEE.SALARY,
    SALARY_GRADE.GRADE_ID,
    EMPLOYEE.DEPARTMENT_ID,
    DEPARTMENT.NAME
FROM EMPLOYEE
    INNER JOIN DEPARTMENT ON EMPLOYEE.DEPARTMENT_ID = DEPARTMENT.DEPARTMENT_ID
    INNER JOIN SALARY_GRADE ON EMPLOYEE.SALARY BETWEEN SALARY_GRADE.LOWER_BOUND AND SALARY_GRADE.UPPER_BOUND
    INNER JOIN JOB ON EMPLOYEE.JOB_ID = JOB.JOB_ID
WHERE JOB.FUNCTION <> 'CLERK'
ORDER BY EMPLOYEE.SALARY DESC;

-- 13 Uzduotis
SELECT
    EMPLOYEE.FIRST_NAME,
    JOB.FUNCTION,
    (SALARY*12) AS ANNUAL_SAL,
    EMPLOYEE.DEPARTMENT_ID,
    DEPARTMENT.NAME,
    SALARY_GRADE.GRADE_ID
FROM EMPLOYEE
    INNER JOIN JOB ON EMPLOYEE.JOB_ID = JOB.JOB_ID
    INNER JOIN SALARY_GRADE ON EMPLOYEE.SALARY BETWEEN SALARY_GRADE.LOWER_BOUND AND SALARY_GRADE.UPPER_BOUND
    INNER JOIN DEPARTMENT ON EMPLOYEE.DEPARTMENT_ID = DEPARTMENT.DEPARTMENT_ID
WHERE (SALARY*12) = 36000 OR
    JOB.FUNCTION = 'CLERK';

    E.FIRST_NAME AS EMPLOYEE,
    E.HIRE_DATE,
    M.FIRST_NAME,
    M.HIRE_DATE
FROM EMPLOYEE E 
    INNER JOIN EMPLOYEE M ON E.MANAGER_ID = M.EMPLOYEE_ID AND E.HIRE_DATE < M.HIRE_DATE;
