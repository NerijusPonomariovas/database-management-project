1. SELECT first_name, name
FROM employee, department
WHERE employee.department_id=department.department_id
ORDER BY name;

2. SELECT first_name, TO_CHAR(hire_date, 'YY-MM-DD') AS hire_date, employee.department_id, name
from employee, department
WHERE employee.department_id=department.department_id;

3. SELECT first_name, TO_CHAR(hire_date, 'YYYY-FMMM-DD'), employee.department_id, name, regional_group
from employee, department, location
WHERE employee.department_id=department.department_id AND
department.location_id=location.location_id AND
(salary+NVL(commission, 0))>1500;

4. SELECT first_name, job_id, salary, grade_id
FROM employee, salary_grade
WHERE employee.salary BETWEEN salary_grade.lower_bound AND salary_grade.upper_bound;

5. SELECT first_name, job_id, salary,
DECODE(grade_id, '1', 'Mazas',
                 '2', 'Didesnis',
                 '3', 'Vidutinis',
                 '4', 'Didelis',
                 '5', 'Didziausias')
FROM employee, salary_grade
WHERE employee.salary BETWEEN salary_grade.lower_bound AND salary_grade.upper_bound;

6. SELECT
    EMPLOYEE.FIRST_NAME,
    EMPLOYEE.JOB_ID,
    EMPLOYEE.SALARY,
    DECODE(SALARY_GRADE.GRADE_ID, 1, 'Mazas', 
				  2, 'Didesnis', 
				  3, 'Vidutinis', 
				  4, 'Didelis', 
 				  5, 'Didziausias')
FROM EMPLOYEE
    INNER JOIN SALARY_GRADE ON
        EMPLOYEE.SALARY BETWEEN SALARY_GRADE.LOWER_BOUND AND SALARY_GRADE.UPPER_BOUND
WHERE DECODE(SALARY_GRADE.GRADE_ID, 1, 'Mazas', 2, 'Didesnis', 3, 'Vidutinis', 4, 'Didelis', 5, 'Didziausias') = INITCAP('&Input');

7. SELECT first_name, TO_CHAR(hire_date, 'YYYY-FMMM-DD'), employee.department_id, name, regional_group
from employee, department, location
WHERE employee.department_id=department.department_id AND
department.location_id=location.location_id 
AND regional_group='DALLAS'

8. SELECT d.department_id, d.name
FROM department d
LEFT JOIN employee e ON d.department_id = e.department_id
WHERE e.department_id IS NULL;

9. SELECT e.first_name EMPLOYEE, e.employee_id EMP_ID,m.first_name MANAGER, m.employee_id EMP_ID
FROM employee e, employee m 
WHERE e.manager_id = m.employee_id;

10. SELECT e.first_name EMPLOYEE, e.employee_id EMP_ID,m.first_name MANAGER, m.employee_id EMP_ID
FROM employee e
LEFT JOIN employee m ON e.manager_id= m.employee_id;

11. SELECT DISTINCT j.function
FROM employee e
JOIN job j ON e.job_id = j.job_id
WHERE e.hire_date BETWEEN TO_DATE('1986-01-01', 'YYYY-MM-DD') AND TO_DATE('1987-06-30', 'YYYY-MM-DD');

12. SELECT
     EMPLOYEE.FIRST_NAME,
     JOB.FUNCTION,
     EMPLOYEE.SALARY,
     SALARY_GRADE.GRADE_ID,
     EMPLOYEE.DEPARTMENT_ID,
     DEPARTMENT.NAME
 FROM EMPLOYEE
     INNER JOIN DEPARTMENT ON EMPLOYEE.DEPARTMENT_ID = DEPARTMENT.DEPARTMENT_ID
     INNER JOIN SALARY_GRADE ON EMPLOYEE.SALARY BETWEEN SALARY_GRADE.LOWER_BOUND AND SALARY_GRADE.UPPER_BOUND
     INNER JOIN JOB ON EMPLOYEE.JOB_ID = JOB.JOB_ID
 WHERE JOB.FUNCTION <> 'CLERK'
 ORDER BY EMPLOYEE.SALARY DESC;

13. SELECT e.first_name, j.function,
(e.salary*12) "ANNUAL_SAL"
from employee e
INNER JOIN job j ON e.job_id=j.job_id
WHERE (e.salary*12)=36000
OR j.function='CLERK'

14.SELECT e.first_name EMPLOYEE,  TO_CHAR(e.hire_date, 'YYYY-FMMM-FMDD'),
m.first_name MANAGER,  TO_CHAR(m.hire_date, 'YYYY-FMMM-FMDD')
FROM employee e, employee m 
WHERE e.manager_id = m.employee_id
AND e.hire_date < m.hire_date;

15. SELECT
     DEPARTMENT.DEPARTMENT_ID,
     DEPARTMENT.NAME

 GROUP BY DEPARTMENT.DEPARTMENT_ID, DEPARTMENT.NAME
