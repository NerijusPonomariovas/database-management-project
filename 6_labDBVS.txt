-- Lentelė: Miestas
CREATE TABLE Miestas (
    MiestoID NUMBER(4) CONSTRAINT MIESTAS_PK PRIMARY KEY,
    Pavadinimas VARCHAR2(100) CONSTRAINT MIESTAS_PAV_NN NOT NULL,
    Salis VARCHAR2(100) DEFAULT 'Lietuva',
    Koordinates VARCHAR2(100)
);

-- Lentelė: Oru simbolis
CREATE TABLE Oru_Simbolis (
    SimbolioID NUMBER(4) CONSTRAINT SIMBOLIS_PK PRIMARY KEY,
    Pavadinimas VARCHAR2(100) CONSTRAINT SIMBOLIS_PAV_NN NOT NULL,
    IkonosURL VARCHAR2(255)
);

-- Lentelė: Oru prognoze
CREATE TABLE Oru_Prognoze (
    PrognozesID NUMBER(5) CONSTRAINT PROGNOZE_PK PRIMARY KEY,
    MiestoID NUMBER(4) CONSTRAINT PROGNOZE_MIESTAS_FK REFERENCES Miestas(MiestoID),
    Auksciausia_Temperatura NUMBER(4,1),
    Zemiausia_Temperatura NUMBER(4,1),
    Vejo_Kryptis VARCHAR2(50),
    UV NUMBER(3,1),
    SimbolioID NUMBER(4) CONSTRAINT PROGNOZE_SIMBOLIS_FK REFERENCES Oru_Simbolis(SimbolioID)
);
