1. SELECT SYSDATE AS "Data" 
FROM DUAL;

2. SELECT employee_id, last_name, salary,
ROUND(salary * 1.15)
FROM employee;

3. SELECT employee_id, last_name, salary,
ROUND(salary * 1.15),
ROUND(salary * 1.15) - salary AS "Skirtumas"
FROM employee;

4. SELECT INITCAP(last_name) AS "Pavarde",
LENGTH(last_name) AS "Pavardės_ilgis"
FROM employee
WHERE SUBSTR(first_name, 1, 1) IN ('A', 'J', 'M');

5. SELECT last_name, MONTHS_BETWEEN(SYSDATE, hire_date) AS "Išdirbti_mėnesiai"
FROM employee;

6. SELECT INITCAP(last_name) ||' per menesi uzdirba '|| TO_CHAR(salary, 'FM$99,999.00')||' bet nori '||
TO_CHAR(salary * 3, 'FM$99,999.00')|| '.' AS "Norimas atlygis"
FROM employee;

7. SELECT INITCAP(last_name) AS "LAST_NAME",
LPAD('$', 10, '$') || salary AS "SALARY"
FROM employee;

8. SELECT last_name,
ROUND(MONTHS_BETWEEN(SYSDATE, hire_date)) AS "Mėnesiai darbe"
FROM employee
ORDER BY 
    ROUND(MONTHS_BETWEEN(SYSDATE, hire_date));

9. SELECT last_name, hire_date,
NEXT_DAY(ADD_MONTHS(hire_date, 4), 'MONDAY') AS "Peržiūros data"
FROM employee
WHERE MONTHS_BETWEEN(SYSDATE, hire_date) > 30;

10. SELECT last_name
FROM employee
WHERE last_name LIKE '%N';

11.SELECT first_name, last_name
FROM employee
WHERE SUBSTR(first_name, 1, 1)=SUBSTR(last_name, -1, 1)
OR 
SUBSTR(last_name, 1, 1)=SUBSTR(first_name, -1, 1);


12. SELECT last_name, MONTHS_BETWEEN(sysdate, hire_date) AS "Isdirbti menesiai", 
ABS(MONTHS_BETWEEN(sysdate, hire_date) - ((sysdate-hire_date)/30)) AS "Apsirikta"
FROM employee;

13. SELECT last_name, hire_date, 
TO_CHAR(NEXT_DAY(ADD_MONTHS(hire_date, 6), 'MONDAY'), 'fmDay", the" Ddspth "of" Month"," RRRR') AS "Peržiūros data"
FROM employee;

14. SELECT last_name, hire_date, 
TO_CHAR(hire_date, 'DAY') as DAY
FROM employee
ORDER BY TO_CHAR(hire_date-1, 'D');

15. SELECT last_name, NVL(TO_CHAR(commission), 'Komisiniai nera nurodyti') AS "Komisiniai"
FROM employee;

16. SELECT INITCAP(first_name) || ' ' || INITCAP(last_name) ||' '|| LPAD('*', ROUND(salary*12/1000)-1,'*') AS "Darbuotojai ir jų atlyginimai"
FROM employee
ORDER BY LPAD('*', ROUND(salary*12/1000)-1,'*') DESC;

17. SELECT DECODE(job_id, 667, 1, 668, 2, 669, 3, 670, 4, 671, 5, 0)
FROM employee;

18. SELECT CASE
WHEN INITCAP(first_name) LIKE 'A%' THEN INITCAP(last_name)||' '|| TO_CHAR(LENGTH(last_name), 'FM99')
WHEN INITCAP(first_name) LIKE 'J%' THEN INITCAP(last_name)||' '|| TO_CHAR(LENGTH(last_name), 'FM99')
WHEN INITCAP(first_name) LIKE 'M%' THEN INITCAP(last_name)||' '|| TO_CHAR(LENGTH(last_name), 'FM99')
END as "case"
FROM employee
WHERE CASE 
WHEN INITCAP(first_name) LIKE 'A%' THEN INITCAP(last_name)||' '|| TO_CHAR(LENGTH(last_name), 'FM99')
WHEN INITCAP(first_name) LIKE 'J%' THEN INITCAP(last_name)||' '|| TO_CHAR(LENGTH(last_name), 'FM99')
WHEN INITCAP(first_name) LIKE 'M%' THEN INITCAP(last_name)||' '|| TO_CHAR(LENGTH(last_name), 'FM99')
END IS NOT NULL;

19. SELECT MIN(salary) AS "SALARY"
FROM employee;

20. SELECT MIN(salary), MAX(salary), AVG(salary)
FROM employee;

21. SELECT job_id, MIN(salary), MAX(salary)
FROM employee
GROUP BY job_id;

22. SELECT COUNT(distinct(manager_id))
FROM employee;

23. SELECT job_id, ROUND(AVG(salary*12), 4) AS "AVG", ROUND(AVG((salary+NVL(commission, 0))*12),4) AS "sal+com"
FROM employee
GROUP BY job_id;

24. SELECT MAX(salary)-MIN(salary)
FROM employee;

25. SELECT department_id, COUNT(employee_id) as "darb. sk."
FROM employee
GROUP BY department_id
HAVING COUNT(employee_id) > 3;

26. SELECT employee_id
FROM employee
GROUP BY employee_id
HAVING COUNT(employee_id)>1;

27. SELECT manager_id, MIN(salary)
FROM employee
GROUP BY manager_id
HAVING MIN(salary)>=1000
ORDER BY MIN(salary);
